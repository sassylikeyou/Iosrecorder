494:    private fun loadLibrary() {
495:        val txtEmpty = viewLibrary.findViewById<TextView>(R.id.txt_empty)
496:        val resolver = contentResolver
497:        val collection = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
498:            android.provider.MediaStore.Video.Media.getContentUri(android.provider.MediaStore.VOLUME_EXTERNAL_PRIMARY)
499:        } else {
500:            android.provider.MediaStore.Video.Media.EXTERNAL_CONTENT_URI
501:        }
502:        val projection = arrayOf(
503:            android.provider.MediaStore.Video.Media._ID,
504:            android.provider.MediaStore.Video.Media.DISPLAY_NAME,
505:            android.provider.MediaStore.Video.Media.DURATION,
506:            android.provider.MediaStore.Video.Media.SIZE
507:        )
508:        val query = resolver.query(
509:            collection,
510:            projection,
511:            android.provider.MediaStore.Video.Media.DATA + " like ? ",
512:            arrayOf("%ScreenRecorder%"),
513:            android.provider.MediaStore.Video.Media.DATE_ADDED + " DESC"
514:        )
515:        val files = mutableListOf<VideoFile>()
516:        val retriever = android.media.MediaMetadataRetriever()
517:        query?.use { cursor ->
518:            val idColumn = cursor.getColumnIndexOrThrow(android.provider.MediaStore.Video.Media._ID)
519:            val nameColumn = cursor.getColumnIndexOrThrow(android.provider.MediaStore.Video.Media.DISPLAY_NAME)
520:            val sizeColumn = cursor.getColumnIndexOrThrow(android.provider.MediaStore.Video.Media.SIZE)
521:            while (cursor.moveToNext()) {
522:                val id = cursor.getLong(idColumn)
523:                val name = cursor.getString(nameColumn)
524:                val size = cursor.getLong(sizeColumn)
525:                val contentUri: Uri = android.content.ContentUris.withAppendedId(android.provider.MediaStore.Video.Media.EXTERNAL_CONTENT_URI, id)
526:                
527:                var thumbnail: android.graphics.Bitmap? = null
